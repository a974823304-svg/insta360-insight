# 个人资料编辑页 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 为已登录用户提供个人资料编辑页（昵称/头像/联系方式/个人简介），资料存后端 SQLite 并与登录账号绑定，仅认证用户可访问，顶栏提供入口。

**Architecture:** 后端在 `users` 表加 4 列（幂等迁移），新增 `GET/PUT /api/user/profile`（JWT 保护组，只改资料列）；前端新增 `ProfileView` 表单路由 + `auth` store 的 fetch/save 能力，顶栏头像下拉加入口。Dev 模式需把真实 admin 身份注入 JWT claims，否则 profile 接口会落到 user 0。

**Tech Stack:** Go + Gin + modernc.org/sqlite（纯 Go 无 CGO）；Vue 3 + Vite + Element Plus + Pinia + Axios；JWT HS256。

## Global Constraints

- **本项目非 git 仓库**：所有 "Commit" 步骤改为「保存文件即可」，严禁 `git init` / `git commit`。
- **Go 工具链在本机网络受限且沙箱无 `go` 命令**：后端 `go build` / `go test` 必须在用户**放行网络环境**执行；本计划的 `[需放行网络]` 步骤由用户跑，助手在本会话只能写代码 + 跑前端验证。
- **纯 Go 无 CGO**：新增 Go 依赖必须纯 Go（已用 `modernc.org/sqlite`）。
- **统一响应约定**：业务失败返回 HTTP 200 + `{ "code": 非0, "message": "..." }`；成功 `{ "code": 0, "data": ... }`。
- **JWT HS256**：资料接口在 `middleware.JWTAuth` 保护组内；从 `c.MustGet("claims").(service.Claims).UserID` 取当前用户。
- **前端 `baseURL: '/api'`**：经 Vite 代理到 `localhost:8080`（见 `vite.config.js`）。
- **前端无单测框架**：验证以 `npx vite build --outDir .build-tmp` 编译通过 + Edge 无头 `--dump-dom` DOM 断言为准（沿用 v15/v16/v17 手段）。
- **主题变量**：`--brand:#3DD9EB` / `--brand-2:#5EA1FF` / `--brand-3:#A07BFF`；玻璃拟态 + 圆角 12px + 动效 `cubic-bezier(0.16,1,0.3,1)`。
- **必填校验（与 spec 一致）**：昵称 + 联系方式必填；头像、个人简介选填。

```go
func (r *UserRepo) migrate() error {
	if _, err := r.db.Exec(`CREATE TABLE IF NOT EXISTS users (
		id            INTEGER PRIMARY KEY AUTOINCREMENT,
		username      TEXT UNIQUE NOT NULL,
		password_hash TEXT NOT NULL,
		role          TEXT NOT NULL DEFAULT 'viewer',
		created_at    DATETIME NOT NULL
	)`); err != nil {
		return err
	}
	have, err := r.columns()
	if err != nil {
		return err
	}
	for _, c := range []string{"nickname TEXT", "avatar TEXT", "contact TEXT", "bio TEXT"} {
		name := strings.Fields(c)[0]
		if have[name] {
			continue
		}
		if _, err := r.db.Exec("ALTER TABLE users ADD COLUMN " + c); err != nil {
			return err
		}
	}
	return nil
}

func (r *UserRepo) columns() (map[string]bool, error) {
	rows, err := r.db.Query("PRAGMA table_info(users)")
	if err != nil {
		return nil, err
	}
	defer rows.Close()
	m := map[string]bool{}
	for rows.Next() {
		var cid int
		var name, ctype string
		var notnull, pk int
		var dflt sql.NullString
		if err := rows.Scan(&cid, &name, &ctype, &notnull, &dflt, &pk); err != nil {
			return nil, err
		}
		m[name] = true
	}
	return m, nil
}
```

```go
func (s *AuthService) UpdateProfile(userID int64, nickname, avatar, contact, bio string) (*model.User, error) {
	if strings.TrimSpace(nickname) == "" {
		return nil, errors.New("昵称为必填项")
	}
	if strings.TrimSpace(contact) == "" {
		return nil, errors.New("联系方式为必填项")
	}
	if len([]rune(bio)) > 500 {
		return nil, errors.New("个人简介不能超过 500 字")
	}
	u, err := s.repo.GetByID(userID)
	if err != nil {
		return nil, err
	}
	u.Nickname = nickname
	u.Avatar = avatar
	u.Contact = contact
	u.Bio = bio
	if err := s.repo.UpdateProfile(u.ID, u); err != nil {
		return nil, err
	}
	return u, nil
}
```

```go
// JWTAuth 解析 Bearer token 并校验;disable 时注入 devUser(本地调试,须为真实账号身份)。
func JWTAuth(v TokenValidator, disable bool, devUser service.Claims) gin.HandlerFunc {
	return func(c *gin.Context) {
		if disable {
			c.Set("claims", devUser)
			c.Next()
			return
		}
		header := c.GetHeader("Authorization")
		if !strings.HasPrefix(header, "Bearer ") {
			c.AbortWithStatusJSON(http.StatusUnauthorized, model.Fail(401, "未提供有效令牌"))
			return
		}
		claims, err := v.ParseToken(strings.TrimPrefix(header, "Bearer "))
		if err != nil {
			c.AbortWithStatusJSON(http.StatusUnauthorized, model.Fail(401, "令牌无效或已过期"))
			return
		}
		c.Set("claims", *claims)
		c.Next()
	}
}
```

```go
// ProfileGet GET /api/user/profile —— 返回当前登录用户资料。
func (h *Auth) ProfileGet(c *gin.Context) {
	claims := c.MustGet("claims").(service.Claims)
	u, err := h.svc.GetProfile(claims.UserID)
	if err != nil {
		c.JSON(http.StatusOK, model.Fail(500, err.Error()))
		return
	}
	c.JSON(http.StatusOK, model.OK(u))
}

// ProfileUpdate PUT /api/user/profile —— 更新昵称/头像/联系方式/简介。
func (h *Auth) ProfileUpdate(c *gin.Context) {
	var q struct {
		Nickname string `json:"nickname"`
		Avatar   string `json:"avatar"`
		Contact  string `json:"contact"`
		Bio      string `json:"bio"`
	}
	if err := c.ShouldBindJSON(&q); err != nil {
		c.JSON(http.StatusOK, model.Fail(400, "请求格式错误"))
		return
	}
	claims := c.MustGet("claims").(service.Claims)
	u, err := h.svc.UpdateProfile(claims.UserID, q.Nickname, q.Avatar, q.Contact, q.Bio)
	if err != nil {
		c.JSON(http.StatusOK, model.Fail(400, err.Error()))
		return
	}
	c.JSON(http.StatusOK, model.OK(u))
}
```

```js
    async fetchProfile() {
      // 资料失败不阻塞主流程,保留已有 user
      try {
        const res = await apiGetProfile()
        if (res && res.code === 0 && res.data) {
          this.user = { ...this.user, ...res.data }
          localStorage.setItem(USER_KEY, JSON.stringify(this.user))
        }
      } catch (e) {
        /* 忽略:无 token / 网络异常时静默 */
      }
    },
    async saveProfile(payload) {
      const res = await apiUpdateProfile(payload)
      if (!res || res.code !== 0) {
        throw new Error((res && res.message) || '保存失败')
      }
      this.user = { ...this.user, ...(res.data || {}) }
      localStorage.setItem(USER_KEY, JSON.stringify(this.user))
      return this.user
    },
```
